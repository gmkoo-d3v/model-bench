    if (interior) {
      interiorKeys.add(x + ',' + z);
      removeVoxel(x, -1, z);

function buildKoiData(rng) {
  const fishCount = 2 + (rng() < 0.5 ? 0 : 1);
  const fish = [];
  const segments = [];
  const scales = [0.7, 0.8, 0.65, 0.5, 0.35];

  for (let fishIndex = 0; fishIndex < fishCount; fishIndex++) {
    const cx = layout.pond.centroid.x + (rng() * 1.6 - 0.8);
    const cz = layout.pond.centroid.z + (rng() * 1.6 - 0.8);

    const maxA = Math.max(1.8, layout.pond.rx - 2.2);
    const a = 1.8 + rng() * Math.max(0, maxA - 1.8);

    const maxB = Math.max(2.5, layout.pond.rz - 3);
    const b = 2.5 + rng() * Math.max(0, maxB - 2.5);

    let loopA = a;
    let loopB = b;
    let validLoop = false;

    for (let attempt = 0; attempt < 14; attempt++) {
      validLoop = true;
      for (let sample = 0; sample < 64; sample++) {
        const phase = sample / 64 * TAU;
        const sampleX = Math.round(cx + loopA * Math.cos(phase));
        const sampleZ = Math.round(cz + loopB * Math.sin(phase));
        if (!layout.pond.interior.has(sampleX + ',' + sampleZ)) {
          validLoop = false;
          break;
        }
      }
      if (validLoop) {
        break;
      }
      loopA = Math.max(1.0, loopA * 0.88);
      loopB = Math.max(1.2, loopB * 0.88);
    }

    if (!validLoop) {
      loopA = Math.max(1.0, loopA);
      loopB = Math.max(1.2, loopB);
    }

    const omega = 0.25 + rng() * 0.15;
    const direction = rng() < 0.5 ? 1 : -1;
    const phase0 = rng() * TAU;
    const whiteIndex = 1 + Math.floor(rng() * 3);

    fish.push({
      cx,
      cz,
      a: loopA,
      b: loopB,
      omega,
      direction,
      phase0
    });

    for (let segmentIndex = 0; segmentIndex < 5; segmentIndex++) {
      const color = segmentIndex === whiteIndex
        ? KOI_WHITE
        : KOI_COLORS[Math.floor(rng() * KOI_COLORS.length)];

      segments.push({
        fish: fishIndex,
        scale: scales[segmentIndex],
        color,
        delay: segmentIndex * 0.22
      });
    }
  }

  layout.koi = { fish, segments };
}
